def need(ok, why):
    if not ok:
        raise ValueError(why)

def sha(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(1 << 20), b""):
            h.update(block)
    return h.hexdigest()

def persist(path, value):
    raw = (json.dumps(value, indent=2, sort_keys=True, allow_nan=False) + "\n").encode()
    with path.open("xb") as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())

def package_outputs(out):
    """Retain complete or partial owned evidence for root's bounded offhost puller."""
    receipt = out.with_suffix(".OFFHOST-PACKAGE.json")
    if receipt.exists():
        return
    archive = out.with_suffix(".PRIVATE-evidence.tar.gz")
    with tarfile.open(archive, "x:gz") as packed:
        for path in sorted(out.rglob("*")):
            if path.is_file() and path.name != "selected-image.docker.tar":
                packed.add(
                    path, arcname=path.relative_to(out).as_posix(), recursive=False
                )
    need(archive.stat().st_size <= 64 << 20, "small offhost package exceeded64MiB")
    persist(
        receipt,
        {
            "path": str(archive),
            "bytes": archive.stat().st_size,
            "sha256": sha(archive),
            "private_not_for_publication": True,
            "root_must_verify_offhost_before_GPU_deletion": True,
        },
    )

def run(argv, timeout, *, stdout=None):
    return subprocess.run(
        argv,
        check=True,
        stdin=subprocess.DEVNULL,
        stdout=stdout or subprocess.PIPE,
        stderr=subprocess.PIPE,
        timeout=timeout,
    )

def verify_model(model, target_manifest):
    """Verify the exact29 staged target files without downloading anything."""
    need(
        sha(target_manifest)
        == "c1309be8991199c9ca5897874ff6e1c056856789e2ef58d129581083dac357c1",
        "frozen target inventory differs",
    )
    inventory = json.loads(Path(target_manifest).read_bytes())
    need(
        inventory["repository"] == "Qwen/Qwen3.6-27B"
        and inventory["revision"] == "6a9e13bd6fc8f0983b9b99948120bc37f49c13e9",
        "exact protected target revision required",
    )
    count = total = 0
    model = Path(model)
    for name, row in inventory["files"].items():
        path = model / name
        need(
            not Path(name).is_absolute()
            and ".." not in Path(name).parts
            and not path.is_symlink()
            and path.is_file()
            and path.stat().st_size == row["size"],
            "target path/size differs",
        )
        h = hashlib.sha256() if "sha256" in row else hashlib.sha1()
        if "sha256" not in row:
            h.update(b"blob " + str(row["size"]).encode() + b"\0")
        with path.open("rb") as stream:
            for block in iter(lambda: stream.read(8 << 20), b""):
                h.update(block)
        need(
            h.hexdigest() == row.get("sha256", row.get("git_blob_sha1")),
            "target bytes differ: " + name,
        )
        count += 1
        total += row["size"]
    need(count == 29 and total == 55586107940, "complete exact target required")
    return {
        "files": count,
        "bytes": total,
        "repository": inventory["repository"],
        "revision": inventory["revision"],
        "downloaded_by_this_job": False,
    }

def layer_blob_identities(layers):
    """Content-addressed identity; retain non-content annotations in raw exports."""
    need(isinstance(layers, list) and len(layers) == 23, "exact23 layer descriptors required")
    result = []
    for layer in layers:
        need(
            isinstance(layer, dict)
            and isinstance(layer.get("mediaType"), str)
            and isinstance(layer.get("digest"), str)
            and type(layer.get("size")) is int
            and layer["size"] >= 0,
            "complete immutable layer descriptor required",
        )
        result.append((layer["mediaType"], layer["digest"], layer["size"]))
    return result

def verify_export_graph(repo, graph, dockerfile):
    """Bind actual exported config/manifest to the already-public23-layer graph."""
    from abstract_harness.configuration_delivery import dockerfile_runtime

    reference = (
        repo
        / "tasks/harness-design/harness/reports/live-mi355x-20261028/controller-neighbors/batch4/selected-metadata"
    )
    need(
        sha(reference / "manifest.json")
        == "84d13e8e4f6777ec4522d701d86e16181c0b170602a7e81db17dab7df525cabb",
        "known-public reference manifest differs",
    )
    need(
        sha(reference / "config.json")
        == "1a19b238c6ce1448fed276005ff14e3425e845db266141deb5391c6a84159586",
        "known-public reference config differs",
    )
    manifest = json.loads((graph / "manifest.json").read_bytes())
    config = json.loads((graph / "config.json").read_bytes())
    prior_manifest = json.loads((reference / "manifest.json").read_bytes())
    prior_config = json.loads((reference / "config.json").read_bytes())
    need(
        layer_blob_identities(manifest["layers"])
        == layer_blob_identities(prior_manifest["layers"]),
        "all23 inherited layer mediaType/digest/size identities must agree",
    )
    need(
        config["rootfs"] == prior_config["rootfs"]
        and len(config["rootfs"]["diff_ids"]) == 23,
        "all23 inherited filesystem diffIDs must agree",
    )
    descriptor = manifest["config"]
    need(
        descriptor["digest"] == "sha256:" + sha(graph / "config.json")
        and descriptor["size"] == (graph / "config.json").stat().st_size,
        "actual raw config descriptor differs",
    )
    need(
        config["os"] == "linux" and config["architecture"] == "amd64",
        "exact linux/amd64 graph required",
    )
    # ArgsEscaped is deprecated Docker Windows-only compatibility metadata.
    # Compare private copies for the verified Linux image; raw config bytes,
    # descriptors, image IDs and archived metadata remain unchanged.
    runtime_config, runtime_prior = copy.deepcopy(config), copy.deepcopy(prior_config)
    need(runtime_prior["os"] == "linux", "Linux comparison reference required")
    args_escaped_values = []
    for compared in (runtime_config, runtime_prior):
        value = compared["config"].get("ArgsEscaped")
        need(value is None or type(value) is bool, "ArgsEscaped must be null/boolean")
        args_escaped_values.append(value)
        compared["config"].pop("ArgsEscaped", None)
    runtime = dockerfile_runtime(dockerfile.read_bytes(), runtime_config, runtime_prior)
    return {
        "status": "EXACT_SELECTED_GRAPH_AND_PUBLIC_INHERITANCE_VERIFIED",
        "manifest": "sha256:" + sha(graph / "manifest.json"),
        "config": descriptor["digest"],
        "inherited_layers": 23,
        "inherited_layer_identity": "mediaType/digest/size; annotations preserved in raw manifests",
        "inherited_diffIDs": 23,
        "runtime": runtime,
        "Linux_comparison_only_ignored_ArgsEscaped": {
            "actual": args_escaped_values[0], "reference": args_escaped_values[1],
            "raw_config_bytes_unchanged": True,
            "basis": "OCI config.md: deprecated compatibility metadata for Docker Windows images",
        },
        "public_reference": BEST_EXISTING,
        "limitation": "This is local graph identity against prior completed public-layer proof; no new registry publication or anonymous pull is claimed.",
    }


def load_runtime(repo, site):
    os.environ["PYTHONDONTWRITEBYTECODE"] = "1"
    sys.dont_write_bytecode = True

    from abstract_harness.engines import LlmdEngine

def current_compatibility(repo, target_manifest, engine, tree_sha, artifact):
    here = repo / "tasks/harness-design/harness"
    lock = json.loads((repo / "baseline/lock.json").read_bytes())
    inventory = json.loads(Path(target_manifest).read_bytes())
    immutable = lock["image"].split("@", 1)[1]
    need(immutable == IMMUTABLE_BASE_DIGEST, "protected immutable FROM digest differs")
    profile = json.loads((repo / "agent-harness/profiles.json").read_bytes())["profiles"]["harness-design"]
    model_runtime = {"engine": "llmd", "model_repository": inventory["repository"],
        "model_revision": inventory["revision"], "model_inventory_sha256": sha(target_manifest),
        "model_args": list(engine.spec.model_args)}
    return artifact.compatibility_binding(context_dir=repo, source_commit=SOURCE,
        harness_tree_sha256=tree_sha(here.parent), base_image_digest=immutable,

        benchmark=(here / "development-benchmark.llmd.json").read_bytes(),
        baseline_lock=(repo / "baseline/lock.json").read_bytes(), model_runtime=model_runtime)

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--config", type=Path, required=True)
    parser.add_argument("--config-sha256", required=True)
    parser.add_argument("--execute", action="store_true")
    args = parser.parse_args()
    need(sha(args.config) == args.config_sha256, "sealed config digest differs")
    cfg = json.loads(args.config.read_bytes())
    if not args.execute:
        print(json.dumps({"prepared_only": True, "GPU_model_provider_actions": 0, "API_capability": 0}))
        return 0
    need(sys.platform == "linux" and sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode,
         "isolated bytecode-free native Linux required")
    try:
        return execute(cfg)
    except Exception as exc:
        out = Path(cfg["output_parent"]) / cfg["run_label"]
        if out.is_dir() and not (out / "FAILED.json").exists():
            persist(out / "FAILED.json", {"status": "FAILED_NO_PUBLIC_SELECTION_CHANGE", "type": type(exc).__name__,
                "reason": str(exc), "finished_utc": datetime.now(timezone.utc).isoformat(), "existing_public_selected_unchanged": BEST_EXISTING})
            package_outputs(out)
        raise


if __name__ == "__main__":
    raise SystemExit(main())
