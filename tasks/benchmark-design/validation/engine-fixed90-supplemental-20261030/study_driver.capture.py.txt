def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def keep(path, value):
    with path.open("x") as stream:
        json.dump(value, stream, indent=2)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())


def verify_source(root):
    manifest = json.loads((HERE / "SOURCE-PINS.json").read_bytes())
    for name, row in manifest["files"].items():
        file = root / name
        if (
            not file.is_file()
            or file.is_symlink()
            or file.stat().st_size != row["bytes"]
            or sha(file) != row["sha256"]
        ):
            raise ValueError("actual96a9 source byte pin differs: " + name)
    return manifest


def summarize(source, out, tokenizer, launch_mono=None):
    suite = source / "tasks/benchmark-design/suite"
    sys.path[:0] = [str(suite), str(source)]
    from bench.gates import judge
    from bench.stats import difference_ci

    plan = json.loads((out / "workload.study.json").read_bytes())
    run = json.loads((out / "events/run.json").read_bytes())
    requests = [
        json.loads(line) for line in (out / "events/requests.jsonl").read_text().splitlines()
    ]
    rows, groups = [], {}
    for raw in requests:
        verdict = judge(raw, plan, [], tokenizer)
        row = {
            "request_id": verdict.request_id,
            "phase": raw["phase"],
            "cell_id": raw["cell_id"],
            "category": verdict.category,
            "reasons": verdict.reasons,
            "TTFT_s": verdict.ttft_s,
        }
        rows.append(row)
        if raw["phase"] == "measure" and verdict.category == "completed":
            groups.setdefault(raw["cell_id"], []).append(verdict.ttft_s)
    controls = groups.get("control_a", []) + groups.get("control_b", [])
    treatment = groups.get("delay100", [])
    difference = difference_ci(
        controls, treatment, statistics.median, resamples=2000, confidence=0.95, seed=2026102903
    )
    proxies = [json.loads(p.read_bytes()) for p in sorted((out / "proxy").glob("*/result.json"))]
    startup = next((x for x in run["startup"] if x["status"] == "valid"), None)
    result = {
        "kind": "SEPARATE_ENGINE_BACKED_POINT_DIAGNOSTIC",
        "official_grade": False,
        "power_or_promotion_claimed": False,
        "all_request_verdicts": rows,
        "groups_TTFT_samples_s": groups,
        "delay_minus_clean_median_descriptive_CI95": difference,
        "healthy_b_minus_a_median_descriptive_CI95":difference_ci(groups.get("control_a",[]),groups.get("control_b",[]),statistics.median,resamples=2000,confidence=.95,seed=2026103004),
        "CI_qualification": "Fixed30perarm, independent-request descriptive percentile bootstrap only; "
        "no measured power, population coverage or regression grade.",
        "client_integrity": run["client_integrity"],
        "proxy_requests": proxies,
        "original_recorder_startup": run["startup"],
        "wrapper_owned_cold_first_valid_generation_s": startup["first_valid_generation_at"]
        - launch_mono
        if startup and launch_mono is not None
        else None,
        "cold_qualification": "Separate physical wrapper launch clock; recorder cold=false "
        "is retained unchanged. No host page-cache-cold claim.",
    }
    keep(out / "study-summary.json", result)
    return result


def require_complete_study(result, summary, report, timing, plan, workload):
    """Refuse a success label for aborted, unchecked or partially offered studies."""
    if (
        result.get("runner_exit_code") != 0
        or result.get("evaluate.py_exit_code") not in {0, 3}
        or result.get("audit_timing.py_exit_code") != 0
        or report.get("status") not in {"valid", "inconclusive"}
        or timing.get("status") != "PASS"
    ):
        raise ValueError("runner or evidence checks did not complete successfully")
    expected = dict(plan["arms_in_order"])
    rows = [row for row in summary["all_request_verdicts"] if row["phase"] == "measure"]
    totals = report["population"]["totals"]
    if (
        totals.get("planned") != sum(expected.values())
        or totals.get("offered") != sum(expected.values())
        or totals.get("not_offered") != 0
        or totals.get("evidence_invalid") != 0
        or Counter(row["cell_id"] for row in rows) != expected
        or len({row["request_id"] for row in rows}) != len(rows)
        or any(row["category"] not in {"completed", "failed", "unfinished"} for row in rows)
        or not any(probe["status"] == "valid" for probe in summary["original_recorder_startup"])
    ):
        raise ValueError("planned study population or valid startup generation is missing")

    if list(expected.items()) != [('control_a',30),('delay100',30),('control_b',30)]:
        raise ValueError('fixed exact30/30/30 protocol required')
    if [row['cell_id'] for row in rows] != ['control_a']*30+['delay100']*30+['control_b']*30:
        raise ValueError('fixed healthyA/delay100/healthyB order required')
    if any(row['category']!='completed' for row in rows) or totals.get('completed')!=90:
        raise ValueError('all90 original measured requests must complete validly')
    integrity=summary['client_integrity'];lag=integrity.get('event_loop_lag_max_s')
    if (workload['client_integrity']['max_event_loop_lag_s']!=.05
        or type(integrity.get('samples')) is not int or integrity['samples']<=0
        or type(lag) not in (int,float) or not math.isfinite(lag) or not 0<=lag<=.05):
        raise ValueError('current measured-window recorder lag must be sampled and<=frozen50ms')


def require_raw_fixed90(out,summary):
    rows=[r for r in summary['all_request_verdicts'] if r['phase']=='measure']
    expected={r['request_id']:r['cell_id'] for r in rows};seen={};delays=0
    for path in sorted((out/'proxy').glob('*/result.json')):
        result=json.loads(path.read_bytes());rid=result['request_id']
        if rid not in expected:continue # Extra original startup attempts remain retained unchanged.
        if rid in seen:raise ValueError('duplicate measured proxy request')
        if result.get('HTTP_status')!=200 or not result.get('upstream_complete') or not result.get('upstream_HTTP_EOF'):
            raise ValueError('every measured request requires complete200/DONE/EOF proxy evidence')
        up=(path.parent/'upstream.sse').read_bytes();down=(path.parent/'downstream.sse').read_bytes()
        if up!=down or not up or b'data: [DONE]' not in up:
            raise ValueError('fixed90 retains exact unchanged complete upstream/downstream SSE')
        for name,raw in [('upstream.sse',up),('downstream.sse',down),('events.jsonl',(path.parent/'events.jsonl').read_bytes())]:
            if result[name]!={'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}:
                raise ValueError('proxy exact raw body/event hash differs')
        events=[json.loads(x) for x in (path.parent/'events.jsonl').read_bytes().splitlines()]
        if expected[rid]=='delay100':
            delays+=1
            if (result['arm']!='delay100' or result.get('injection_applied') is not True
                or result['delay_end_ns']-result['delay_begin_ns']<100000000
                or result['first_visible_downstream_write_ns']<result['delay_end_ns']):
                raise ValueError('all30 delays must be actual100ms before first visible downstream')
            for name,key in [('delay_begin','delay_begin_ns'),('delay_end','delay_end_ns')]:
                if len([e for e in events if e['event']==name and e['monotonic_ns']==result[key]])!=1:
                    raise ValueError('exact delay event provenance missing')
        elif result['arm']!='clean' or result.get('injection_applied'):
            raise ValueError('healthy arms must retain unchanged clean streams')
        seen[rid]=result
    if set(seen)!=set(expected) or len(seen)!=90 or delays!=30:
        raise ValueError('exact90 complete proxy streams and30 injected delays required')
    return {'status':'PASS_FIXED90_EXACT_RAW_SSE','measured_streams':90,'actual100ms_delays':30,'all_bytes_identical':True}


def admit_fixed90(work_stop,delete_by,wall,mono):
    if (not all(math.isfinite(x) for x in [work_stop,delete_by,wall,mono])
        or delete_by-work_stop!=90 or work_stop-wall<180 or delete_by-wall<240):
        raise ValueError('skip: new contract needs180usablework and240toDELETE with unchanged90gap')
    return mono+180

def export(out, destination):
    if destination.exists() or destination.is_symlink():
        raise ValueError("new archive required")
    allowed = {
        "STUDY-PLAN.json",
        "SOURCE-PINS.json",
        "workload.study.json",
        "ownership.json",
        "engine-command.json",
        "launch.json",
        "engine.log",
        "runner.log",
        "report.json",
        "summary.md",
        "evaluate.log",
        "independent-timing.json",
        "audit.log",
        "study-summary.json",
        "STUDY-RESULT.json",
        "FIXED90-INTEGRITY.json",
        "RECOVERY-RESULT.json",
        "events/run.json",
        "events/requests.jsonl",
    }
    files = []
    for path in out.rglob("*"):
        if path.is_symlink():
            raise ValueError("study evidence symlink refused")
        if not path.is_file():
            continue
        rel = path.relative_to(out).as_posix()
        proxy = (
            len(Path(rel).parts) == 3
            and rel.startswith("proxy/")
            and path.name in {"upstream.sse", "downstream.sse", "events.jsonl", "result.json"}
        )
        if rel not in allowed and not proxy:
            raise ValueError("unexpected study evidence file: " + rel)
        files.append((rel, path))
    if len(files) > 512 or sum(p.stat().st_size for _, p in files) > 64 << 20:
        raise ValueError("bounded small diagnostic evidence")
    manifest = {name: {"bytes": path.stat().st_size, "sha256": sha(path)} for name, path in files}
    with destination.open("xb") as raw, tarfile.open(fileobj=raw, mode="w:gz") as archive:
        for name, path in files:
            info = tarfile.TarInfo(name)
            info.size, info.mode = path.stat().st_size, 0o600
            with path.open("rb") as stream:
                archive.addfile(info, stream)
        body = (json.dumps(manifest, indent=2) + "\n").encode()
        info = tarfile.TarInfo("export-manifest.json")
        info.size, info.mode = len(body), 0o600
        archive.addfile(info, io.BytesIO(body))
    verified = verify_export(destination)
    if verified["files"] != manifest:
        raise ValueError("independent diagnostic archive readback differs")
    return {
        "archive": str(destination),
        "bytes": destination.stat().st_size,
        "sha256": sha(destination),
        **verified,
    }


def verify_export(archive_path):
    actual, declared, seen, total = {}, None, set(), 0
    with tarfile.open(archive_path, "r|gz") as archive:
        for member in archive:
            if (
                not member.isfile()
                or member.name in seen
                or member.name.startswith("/")
                or ".." in Path(member.name).parts
                or member.size < 0
            ):
                raise ValueError("unsafe diagnostic archive member")
            seen.add(member.name)
            total += member.size
            if total > (65 << 20):
                raise ValueError("bounded diagnostic archive")
            stream = archive.extractfile(member)
            if member.name == "export-manifest.json":
                if member.size > 1 << 20:
                    raise ValueError("bounded diagnostic manifest")
                declared = json.loads(stream.read())
            else:
                value, size = hashlib.sha256(), 0
                while block := stream.read(1 << 20):
                    value.update(block)
                    size += len(block)
                actual[member.name] = {"bytes": size, "sha256": value.hexdigest()}
    if not actual or declared != actual:
        raise ValueError("complete diagnostic archive body/manifest mismatch")
    return {"files": actual, "verified_complete_archive_manifest": True}
